// src/orders/refund.ts
import { randomUUID } from "node:crypto";
import { db } from "../db/client.js";
import { releaseStock } from "../inventory/stock.js";
import { eventBus } from "../events/bus.js";
import type { PaymentTransaction } from "../db/schema.js";

export async function refundOrder(params: {
  tenantId: string;
  orderId: string;
  reason: string;
  idempotencyKey: string;
}): Promise<{
  success: boolean;
  isDuplicate?: boolean;
  error?: string;
  refundPaymentId?: string;
}> {
  const { tenantId, orderId, reason, idempotencyKey } = params;

  const existing = db.getPaymentByIdempotency(tenantId, idempotencyKey);
  if (existing) {
    return {
      success: true,
      isDuplicate: true,
      refundPaymentId: existing.id
    };
  }

  const order = db.getOrder(orderId);
  if (!order || order.tenantId !== tenantId) {
    return {
      success: false,
      error: "Order not found"
    };
  }
  if (order.status !== "paid") {
    return {
      success: false,
      error: "Order is not paid"
    };
  }

  for (const item of order.items) {
    await releaseStock(tenantId, item.sku, item.quantity);
  }

  order.status = "refunded";
  order.updatedAt = new Date().toISOString();
  db.saveOrder(order);

  const payment: PaymentTransaction = {
    id: randomUUID(),
    tenantId,
    orderId,
    amountCents: order.totalAmountCents,
    status: "refunded",
    idempotencyKey,
    createdAt: new Date().toISOString()
  };
  db.savePayment(payment);
  db.registerIdempotencyKey(tenantId, idempotencyKey, payment.id);

  await eventBus.publish({
    id: randomUUID(),
    type: "order.refunded",
    tenantId,
    payload: {
      orderId,
      reason,
      amountCents: order.totalAmountCents
    },
    timestamp: new Date().toISOString()
  });

  return {
    success: true,
    isDuplicate: false,
    refundPaymentId: payment.id
  };
}
